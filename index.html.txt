        let selectedFilter = 'all';

        let qrCodeInstance = null;
        let activeQRUrl = "";

        // Render Cards
        function renderCircuits() {
            const grid = document.getElementById('circuitsGrid');
            const searchVal = document.getElementById('searchInput').value.toLowerCase();
            grid.innerHTML = '';

            const filtered = circuitsData.filter(item => {
                const matchesFilter = selectedFilter === 'all' || item.category === selectedFilter;
                const matchesSearch = item.name.toLowerCase().includes(searchVal) || 
                                      item.components.some(c => c.toLowerCase().includes(searchVal)) ||
                                      item.tag.toLowerCase().includes(searchVal);
                return matchesFilter && matchesSearch;
            });

            if(filtered.length === 0) {
                grid.innerHTML = `<div class="col-span-full text-center py-12 text-slate-500">No circuits found matching your query.</div>`;
                return;
            }

            filtered.forEach(c => {
                const card = document.createElement('div');
                card.className = "bg-slate-900 border border-slate-800 hover:border-cyan-500/40 rounded-xl p-5 flex flex-col justify-between transition-all duration-300 hover:shadow-lg hover:shadow-cyan-500/5 group";
                
                card.innerHTML = `
                    <div>
                        <div class="flex justify-between items-start mb-3">
                            <span class="text-[10px] font-bold uppercase tracking-wider text-cyan-400 bg-cyan-950/80 border border-cyan-500/30 px-2 py-0.5 rounded-full">${c.tag}</span>
                            <button onclick="openSingleQR(${c.id})" title="Generate QR Code" class="text-slate-500 hover:text-cyan-400 transition text-sm">
                                <i class="fa-solid fa-qrcode"></i>
                            </button>
                        </div>
                        <h2 class="text-lg font-bold text-slate-100 group-hover:text-cyan-400 transition-colors mb-4">${c.name}</h2>
                        
                        <!-- Key Components -->
                        <div class="mb-4">
                            <h4 class="text-xs font-semibold text-slate-400 uppercase tracking-wider mb-2 flex items-center gap-1.5">
                                <i class="fa-solid fa-layer-group text-cyan-500"></i> Key Components
                            </h4>
                            <ul class="text-xs text-slate-300 space-y-1 list-disc list-inside bg-slate-950/60 p-2.5 rounded-lg border border-slate-800/80">
                                ${c.components.map(comp => `<li>${comp}</li>`).join('')}
                            </ul>
                        </div>

                        <!-- Wiring Summary -->
                        <div class="mb-6">
                            <h4 class="text-xs font-semibold text-slate-400 uppercase tracking-wider mb-2 flex items-center gap-1.5">
                                <i class="fa-solid fa-plug text-cyan-500"></i> Wiring Summary
                            </h4>
                            <ul class="text-xs text-slate-300 space-y-1 bg-slate-950/60 p-2.5 rounded-lg border border-slate-800/80">
                                ${c.summary.map(s => `<li class="border-b border-slate-800/50 pb-1 last:border-none">${s}</li>`).join('')}
                            </ul>
                        </div>
                    </div>

                    <div class="pt-2">
                        <button onclick="openBuyModal(${c.id})" class="w-full py-2 bg-gradient-to-r from-cyan-500/20 to-blue-500/20 hover:from-cyan-500 hover:to-blue-600 text-cyan-300 hover:text-slate-950 border border-cyan-500/40 rounded-lg text-xs font-bold transition duration-200 flex items-center justify-center gap-2">
                            <i class="fa-solid fa-cart-shopping"></i> Buy / Inquiry Message
                        </button>
                    </div>
                `;
                grid.appendChild(card);
            });
        }

        // Filtering & Search
        function setFilter(category) {
            selectedFilter = category;
            document.querySelectorAll('.filter-btn').forEach(btn => {
                btn.className = "filter-btn px-3 py-1.5 text-xs rounded-lg font-medium border border-slate-800 bg-slate-950 text-slate-400 hover:border-slate-700";
            });
            event.target.className = "filter-btn active px-3 py-1.5 text-xs rounded-lg font-medium border border-cyan-500 bg-cyan-500/20 text-cyan-300";
            renderCircuits();
        }

        function filterCircuits() {
            renderCircuits();
        }

        // Modals Logic
        function openBuyModal(circuitId) {
            const circuit = circuitsData.find(c => c.id === circuitId);
            currentCircuitForInquiry = circuit;
            document.getElementById('modalCircuitTitle').innerText = circuit.name;
            document.getElementById('buyModal').classList.remove('hidden');
        }

        function closeBuyModal() {
            document.getElementById('buyModal').classList.add('hidden');
        }

        function sendInquiry(e) {
            e.preventDefault();
            const name = document.getElementById('buyerName').value;
            const phone = document.getElementById('buyerPhone').value;
            const qty = document.getElementById('buyQuantity').value;
            const note = document.getElementById('buyerNote').value;

            const text = `Hello Sujit Kumar,%0A%0AI want to inquire about buying/building the following circuit:%0A*Circuit:* ${currentCircuitForInquiry.name}%0A*Quantity:* ${qty}%0A*Buyer Name:* ${name}%0A*Phone:* ${phone}%0A*Notes:* ${note}`;
            
            // Redirect to WhatsApp
            window.open(`https://wa.me/?text=${text}`, '_blank');
            closeBuyModal();
        }

        // QR Code Modal
        function generateQRCode(url) {
            const container = document.getElementById('qrcodeCanvas');
            container.innerHTML = '';
            qrCodeInstance = new QRCode(container, {
                text: url,
                width: 180,
                height: 180,
                colorDark: "#020617",
                colorLight: "#ffffff",
                correctLevel: QRCode.CorrectLevel.H
            });
            activeQRUrl = url;
        }

        function openMainQRModal() {
            document.getElementById('qrTitle').innerText = "Sujit Kumar Circuit Store";
            document.getElementById('qrSubtitle').innerText = "Scan to visit full website";
            document.getElementById('qrModal').classList.remove('hidden');
            generateQRCode(window.location.href);
        }

        function openSingleQR(circuitId) {
            const circuit = circuitsData.find(c => c.id === circuitId);
            document.getElementById('qrTitle').innerText = circuit.name;
            document.getElementById('qrSubtitle').innerText = "Scan to view circuit details & components";
            document.getElementById('qrModal').classList.remove('hidden');
            
            const circuitUrl = `${window.location.origin}${window.location.pathname}?circuit=${circuit.id}`;
            generateQRCode(circuitUrl);
        }

        function closeQRModal() {
            document.getElementById('qrModal').classList.add('hidden');
        }

        function copyQRLink() {
            navigator.clipboard.writeText(activeQRUrl);
            alert("Link copied to clipboard!");
        }
